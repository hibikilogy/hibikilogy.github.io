{# i18n helper — site translations override theme i18n defaults. Throws when the key is missing everywhere (lookup chain below). Call: {{}} #} {% component i18n.t(key: string, lang: string) %} {%- set i18n_path = "themes/hibikilogy/i18n/" ~ lang ~ ".toml" -%} {%- set theme_i18n = load_data(path=i18n_path, required=false) -%} {%- set site_config = load_data(path="zola.toml", required=false) -%} {%- if not site_config -%} {%- set site_config = load_data(path="config.toml", required=false) -%} {%- endif -%} {# 1. Site per-language translation #} {%- if site_config and site_config.languages is defined and site_config.languages[lang] is defined and site_config.languages[lang].translations is defined and site_config.languages[lang].translations[key] is defined -%} {{- site_config.languages[lang].translations[key] -}} {# 2. Top-level [translations] belongs to the default language only #} {%- elif site_config and lang == site_config.default_language and site_config.translations is defined and site_config.translations[key] is defined -%} {{- site_config.translations[key] -}} {# 3. Theme i18n baseline #} {%- elif theme_i18n and theme_i18n[key] is defined -%} {{- theme_i18n[key] -}} {# 4. No fallback: fail the build when the key is missing everywhere #} {%- else -%} {{- throw(message="Translation key '" ~ key ~ "' for language '" ~ lang ~ "' is missing") -}} {%- endif -%} {% endcomponent i18n.t %}
